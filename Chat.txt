using UnityEngine;

using System.Collections.Generic;



public class AutomaticRobotBuilder : MonoBehaviour

{

    [Header("Automatic Detection")]

    public bool buildOnStart = true;



    [Header("Wheel Detection")]

    public float minimumWheelRadius = 0.01f;

    public float maximumWheelRadius = 0.15f;



    [Header("Physics")]

    public bool addWheelCollider = true;



    private Rigidbody robotRigidbody;



    private readonly List<Transform> wheelCandidates =

        new List<Transform>();



    void Start()

    {

        if (!buildOnStart)

            return;



        BuildRobot();

    }



    public void BuildRobot()

    {

        Debug.Log(

            "========================================"

        );



        Debug.Log(

            "AutomaticRobotBuilder START"

        );



        robotRigidbody =

            GetComponent<Rigidbody>();



        if (robotRigidbody == null)

        {

            robotRigidbody =

                GetComponentInParent<Rigidbody>();

        }



        if (robotRigidbody == null)

        {

            Debug.LogError(

                "AutomaticRobotBuilder: Rigidbodyがありません。"

            );



            return;

        }



        DetectWheels();



        Debug.Log(

            "車輪候補: " +

            wheelCandidates.Count +

            " 個"

        );



        for (

            int i = 0;

            i < wheelCandidates.Count;

            i++

        )

        {

            Debug.Log(

                "Wheel[" +

                i +

                "] = " +

                GetHierarchyPath(

                    wheelCandidates[i]

                )

            );

        }



        Debug.Log(

            "AutomaticRobotBuilder END"

        );



        Debug.Log(

            "========================================"

        );

    }



    void DetectWheels()

    {

        wheelCandidates.Clear();



        Transform[] allTransforms =

            GetComponentsInChildren<Transform>(

                true

            );



        foreach (

            Transform current

            in allTransforms

        )

        {

            if (

                current == transform

            )

            {

                continue;

            }



            if (

                IsWheelName(

                    current.name

                )

            )

            {

                AddWheelCandidate(

                    current

                );



                continue;

            }



            MeshRenderer renderer =

                current.GetComponent<MeshRenderer>();



            if (renderer == null)

            {

                continue;

            }



            Bounds bounds =

                renderer.bounds;



            Vector3 size =

                bounds.size;



            float min =

                Mathf.Min(

                    size.x,

                    Mathf.Min(

                        size.y,

                        size.z

                    )

                );



            float max =

                Mathf.Max(

                    size.x,

                    Mathf.Max(

                        size.y,

                        size.z

                    )

                );



            // 円盤・車輪らしい形状

            float ratio =

                min / Mathf.Max(

                    max,

                    0.0001f

                );



            if (

                ratio < 0.9f

            )

            {

                continue;

            }



            float radius =

                max * 0.5f;



            if (

                radius <

                minimumWheelRadius ||

                radius >

                maximumWheelRadius

            )

            {

                continue;

            }



            AddWheelCandidate(

                current

            );

        }

    }



    bool IsWheelName(

        string objectName

    )

    {

        string name =

            objectName.ToLower();



        return

            name.Contains("wheel") ||

            name.Contains("tire") ||

            name.Contains("tyre") ||

            name.Contains("ホイール") ||

            name.Contains("タイヤ") ||

            name.Contains("車輪");

    }



    void AddWheelCandidate(

        Transform candidate

    )

    {

        if (

            candidate == null

        )

        {

            return;

        }



        if (

            wheelCandidates.Contains(

                candidate

            )

        )

        {

            return;

        }



        wheelCandidates.Add(

            candidate

        );

    }



    string GetHierarchyPath(

        Transform target

    )

    {

        string path =

            target.name;



        Transform parent =

            target.parent;



        while (

            parent != null &&

            parent != transform

        )

        {

            path =

                parent.name +

                "/" +

                path;



            parent =

                parent.parent;

        }



        return path;

    }

}

    }



    IEnumerator WaitForRobot()

    {

        Debug.Log(

            "[AutoRobot] Robot生成待機開始"

        );



        while (!built)

        {

            GameObject robot =

                FindRobot();



            if (robot != null)

            {

                detectedRobot = robot;



                Debug.Log(

                    "[AutoRobot] Robot検出: " +

                    robot.name

                );



                BuildRobot(

                    robot

                );



                built = true;



                yield break;

            }



            yield return null;

        }

    }



    GameObject FindRobot()

    {

        GameObject robot =

            GameObject.Find(

                robotName

            );



        if (robot != null)

            return robot;



        RobotProgram program =

            FindAnyObjectByType<RobotProgram>();



        if (program != null)

        {

            Transform root =

                program.transform;



            if (root != null)

                return root.gameObject;

        }



        return null;

    }

        GameObject robot

    )

    {

            "[AutoRobot] ======================="

        );

        );

            robot.GetComponent<Rigidbody>();

        {

            rb =

            );

            robot.GetComponent<

            builder =

                robot.AddComponent<

using UnityEngine;



public class ColorSensor : MonoBehaviour

{

    public float sensorHeight = 0.2f;

    public float rayDistance = 2f;



    public Color GetColor()

    {

        Vector3 origin =

            transform.position +

            Vector3.up * sensorHeight;



        RaycastHit[] hits =

            Physics.RaycastAll(

                origin,

                Vector3.down,

                rayDistance

            );



        float closestDistance = Mathf.Infinity;

        Renderer targetRenderer = null;



        foreach (RaycastHit hit in hits)

        {

            if (hit.collider == null)

                continue;



            // ロボット自身を無視

            if (hit.collider.transform.root == transform.root)

                continue;



            // カラーテスト床だけを対象にする

            if (!hit.collider.CompareTag("ColorSensorTarget"))

                continue;



            Renderer renderer =

                hit.collider.GetComponent<Renderer>();



            if (renderer == null)

                continue;



            if (hit.distance < closestDistance)

            {

                closestDistance = hit.distance;

                targetRenderer = renderer;

            }

        }



        if (targetRenderer != null)

        {

            return targetRenderer.material.color;

        }



        return Color.black;

    }

}

    // ==================================================

    // Motor simulation

    void SimulateMotor()

    {

        float delta =

            targetSpeed -

            currentSpeed;



        float acceleration =

            3000f;



        float change =

            acceleration *

            Time.fixedDeltaTime;



        currentSpeed =

            Mathf.MoveTowards(

                currentSpeed,

                targetSpeed,

                change

            );



        float speedRatio =

            Mathf.Abs(targetSpeed) /

            Mathf.Max(

                1f,

                maxSpeed

            );



        currentTorque =

            Mathf.Sign(targetSpeed) *

            maxTorque *

            speedRatio;



        // ------------------------------------------

        // 車輪を実際に回す

        // ------------------------------------------



        float rotation =

            currentSpeed *

            Time.fixedDeltaTime;



        currentAngle += rotation;



        wheel.Rotate(

            rotationAxis,

            rotation,

            Space.Self

        );

    }



    // ==================================================

    // Motor state

    // ==================================================



    void UpdateMotorState()

    {

        if (wheel == null)

            return;



        // 車輪の回転による

        // 仮想的な接地駆動

        //

        // ここでは車体を直接押さない。

        // 将来的にWheelCollider / 接触点から

        // 摩擦力を取得する。



        float wheelAngularVelocity =

            currentSpeed *

            Mathf.Deg2Rad;



        float linearVelocity =

            wheelAngularVelocity *

            wheelRadius;



        Vector3 forward =

            transform.parent.forward;



        forward.y = 0f;



        if (forward.sqrMagnitude >

            0.0001f)

        {

            forward.Normalize();



            Vector3 velocity =

                robotRigidbody.linearVelocity;



            float forwardVelocity =

                Vector3.Dot(

                    velocity,

                    forward

                );



            // 接地している場合だけ

            // 摩擦による駆動を発生させる

            float slip =

                linearVelocity -

                forwardVelocity;



            float frictionForce =

                slip *

                friction;



            frictionForce =

                Mathf.Clamp(

                    frictionForce,

                    -maxTorque / Mathf.Max(

                        wheelRadius,

                        0.001f

                    ),

                    maxTorque / Mathf.Max(

                        wheelRadius,

                        0.001f

                    )

                );



            Vector3 force =

                forward *

                frictionForce;



            rbAddForceAtWheel(

                force

            );

        }

    }



    void rbAddForceAtWheel(

        Vector3 force

    )

    {

        robotRigidbody.AddForceAtPosition(

            force,

            wheel.position,

            ForceMode.Force

        );

    }



    // ==================================================

    // Reset angle

    // ==================================================



    public void ResetAngle()

    {

        currentAngle = 0f;

    }



    public void ResetAngle(

        float angle

    )

    {

        currentAngle = angle;

    }

    void OnDrawGizmosSelected()

    {

                rotationAxis.normalized

            );



        Gizmos.DrawLine(

            wheel.position -

            axis * wheelRadius,

            wheel.position +

            axis * wheelRadius

        );

    }

using UnityEngine;

using System;

using System.IO;

using System.Collections.Concurrent;



public class PythonRunner : MonoBehaviour

{

    public string pythonExecutable = "python";



    private string pythonFilePath;

    private Process pythonProcess;



    private readonly ConcurrentQueue<string> pythonMessages =

        new ConcurrentQueue<string>();



    private RobotAPI robot;



    void Awake()

    {

        pythonFilePath = Path.Combine(

            Application.dataPath,

            "RobotSimulator",

            "PythonRuntime",

            "robot_bridge.py"

        );



        robot = GetComponent<RobotAPI>();



        if (robot == null)

        {

            robot = gameObject.AddComponent<RobotAPI>();

        }

    }



    void Start()

    {

        UnityEngine.Debug.Log(

            "===== PythonRunner Start ====="

        );



        if (!File.Exists(pythonFilePath))

        {

            UnityEngine.Debug.LogError(

                "Pythonファイルが見つかりません: " +

                pythonFilePath

            );



            return;

        }



        StartPython();

    }



    void Update()

    {

        while (

            pythonMessages.TryDequeue(

                out string message

            )

        )

        {

            ProcessPythonMessage(message);

        }

    }



    void StartPython()

    {

        try

        {

            ProcessStartInfo startInfo =

                new ProcessStartInfo();



            startInfo.FileName =

                pythonExecutable;



            startInfo.Arguments =

                "-u \"" +

                pythonFilePath +

                "\"";



            startInfo.UseShellExecute =

                false;



            startInfo.RedirectStandardOutput =

                true;



            startInfo.RedirectStandardError =

                true;



            startInfo.RedirectStandardInput =

                true;



            startInfo.CreateNoWindow =

                true;



            pythonProcess =

                new Process();



            pythonProcess.StartInfo =

                startInfo;



            pythonProcess.OutputDataReceived +=

                OnPythonOutput;



            pythonProcess.ErrorDataReceived +=

                OnPythonError;



            pythonProcess.Start();



            pythonProcess.BeginOutputReadLine();



            pythonProcess.BeginErrorReadLine();



            UnityEngine.Debug.Log(

                "Pythonプロセス起動成功"

            );

        }

        catch (Exception exception)

        {

            UnityEngine.Debug.LogError(

                "Python起動失敗:\n" +

                exception

            );

        }

    }



    void OnPythonOutput(

        object sender,

        DataReceivedEventArgs e

    )

    {

        if (string.IsNullOrEmpty(e.Data))

        {

            return;

        }



        pythonMessages.Enqueue(e.Data);

    }



    void OnPythonError(

        object sender,

        DataReceivedEventArgs e

    )

    {

        if (string.IsNullOrEmpty(e.Data))

        {

            return;

        }



        pythonMessages.Enqueue(

            "PYTHON_ERROR|" + e.Data

        );

    }



    void ProcessPythonMessage(

        string message

    )

    {

        UnityEngine.Debug.Log(

            "Python: " + message

        );



        if (!message.StartsWith("PBX:"))

        {

            return;

        }



        string command =

            message.Substring(4);



        string[] parts =

            command.Split('|');



        if (parts.Length == 0)

        {

            return;

        }



        switch (parts[0])

        {

            case "HUB_CREATE":

                HandleHubCreate(parts);

                break;



            case "MOTOR_CREATE":

                HandleMotorCreate(parts);

                break;



            case "MOTOR_RUN":

                HandleMotorRun(parts);

                break;



            case "MOTOR_STOP":

                HandleMotorStop(parts);

                break;



            case "MOTOR_BRAKE":

                HandleMotorStop(parts);

                break;



            case "MOTOR_HOLD":

                HandleMotorStop(parts);

                break;



            case "MOTOR_RESET_ANGLE":

                HandleMotorResetAngle(parts);

                break;



            default:

                UnityEngine.Debug.LogWarning(

                    "未知のPBXコマンド: " +

                    command

                );

                break;

        }

    }



    void HandleHubCreate(

        string[] parts

    )

    {

        UnityEngine.Debug.Log(

            "Pybricks Hub: " +

            (

                parts.Length > 1

                    ? parts[1]

                    : "Unknown"

            )

        );

    }



    void HandleMotorCreate(

        string[] parts

    )

    {

        if (parts.Length < 2)

        {

            return;

        }



        UnityEngine.Debug.Log(

            "Pybricks Motor接続: Port " +

            parts[1]

        );

    }



    void HandleMotorRun(

        string[] parts

    )

    {

        if (parts.Length < 3)

        {

            return;

        }



        string port =

            parts[1];



        if (!float.TryParse(

            parts[2],

            out float speed

        ))

        {

            UnityEngine.Debug.LogError(

                "Motor速度を解釈できません: " +

                parts[2]

            );



            return;

        }



        // 現段階では

        // Pybricksのdeg/sをUnity側の

        // モーターパワーへ暫定変換。

        float power =

            Mathf.Clamp(

                speed * 0.1f,

                -100f,

                100f

            );



        if (port == "A")

        {

            robot.SetLeftMotor(

                power

            );



            UnityEngine.Debug.Log(

                "Port A → LeftMotor | " +

                "Speed: " +

                speed +

                " | Power: " +

                power

            );

        }

        else if (port == "B")

        {

            robot.SetRightMotor(

                power

            );



            UnityEngine.Debug.Log(

                "Port B → RightMotor | " +

                "Speed: " +

                speed +

                " | Power: " +

                power

            );

        }

        else

        {

            UnityEngine.Debug.LogWarning(

                "未割り当てPort: " +

                port

            );

        }

    }



    void HandleMotorStop(

        string[] parts

    )

    {

        if (parts.Length < 2)

        {

            return;

        }



        string port =

            parts[1];



        if (port == "A")

        {

            robot.SetLeftMotor(

                0f

            );

        }

        else if (port == "B")

        {

            robot.SetRightMotor(

                0f

            );

        }

    }

    )

    {

        if (parts.Length < 3)

        {

        }

            "Port " +

            parts[1] +

            " の角度を " +

            parts[2] +

            " deg に設定"

        );

    }



    public void SendCommand(

    )

    {

        if (

            pythonProcess == null ||

            pythonProcess.HasExited

        )

        {

            UnityEngine.Debug.LogError(

                "Pythonプロセスが動いていません。"

            );



            return;

        }



        pythonProcess.StandardInput.WriteLine(

            command

        );



        pythonProcess.StandardInput.Flush();



        UnityEngine.Debug.Log(

            "Unity → Python: " +

            command

        );

    }

            {

            }

        }

    }
